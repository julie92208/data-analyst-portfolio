Python Code – Data Cleaning

import pandas as pd # Load datasets doctor = pd.read_csv("data-doctor.csv") nurse = pd.read_csv("data-nurse.csv") # Select useful columns cols = ["Location", "ParentLocation", "Period", "Indicator", "Value"] doctor = doctor[cols] nurse = nurse[cols] # Rename for clarity doctor.rename(columns={"Value": "Doctors_per_10k"}, inplace=True) nurse.rename(columns={"Value": "Nurses_per_10k"}, inplace=True) # Merge merged = pd.merge( doctor, nurse, on=["Location", "ParentLocation", "Period"], how="inner" ) # Compute nurse-to-doctor ratio merged["Nurse_to_Doctor_Ratio"] = ( merged["Nurses_per_10k"] / merged["Doctors_per_10k"] ) merged.head()

# 1. Regional average ratio regional_avg = merged.groupby("ParentLocation")["Nurse_to_Doctor_Ratio"].mean() # 2. Top 5 countries top5 = merged.groupby("Location")["Nurse_to_Doctor_Ratio"].mean().nlargest(5) # 3. Canada 2018–2023 canada = merged[merged["Location"] == "Canada"].sort_values("Period") # 4. Correlation (2023) data2023 = merged[merged["Period"] == 2023] correlation = data2023["Doctors_per_10k"].corr(data2023["Nurses_per_10k"])
